export function splitInstallments(total: number, count: number, rate: number) {
const base = rate === 0 ? Math.floor((total / count) * 100) / 100 : pmt(total, count, rate);
const installments = Array.from({ length: count }, () => base);
// the last installment absorbs the cent difference
const diff = Math.round((total - base * count) * 100) / 100;
installments[count - 1] = Math.round((base + diff) * 100) / 100;
return installments;
}
